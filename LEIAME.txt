# 1. Realizar o build da imagem
docker build -t go-rocks .

# 2. Rodar o container
docker run --rm -it -d --name go-rocks -p 8080:8080 go-rocks

# 3. Acessar o site
http://localhost:8080

# 4. Verificar o tamanho da imagem
docker images go-rocks

# 5. Logar no docker hub
docker login

# 6. Criar a tag para a imagem
docker tag go-rocks asoares73/fullcycle_go-rocks:latest

# 7. Enviar a imagem para o Docker Hub
docker push asoares73/fullcycle_go-rocks:latest

# 8. Verificar a imagem disponibilizada no Docker Hub
Acesse: https://hub.docker.com/repository/docker/asoares73/fullcycle_go-rocks/general

# 9. Imagem Pull
docker pull asoares73/fullcycle_go-rocks:latest

# 10. Executar a imagem diretamente do Docker Hub
docker run --rm -it -d --name fullcyle_go-rocks asoares73/fullcycle_go-rocks:latest
docker logs
